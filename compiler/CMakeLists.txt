cmake_minimum_required(VERSION 3.24)
project(ssarv_team VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_options(-Wall -Wextra -Wpedantic)

add_library(ssarv_utilities src/ir_support.cpp src/bitset.cpp src/graph.cpp src/alloc_result.cpp)
target_include_directories(ssarv_utilities PUBLIC include)
add_library(ssarv_analysis src/analysis.cpp)
target_link_libraries(ssarv_analysis PUBLIC ssarv_utilities)
# Analysis symbols are intentionally resolved by either fixture providers
# (isolated allocator tests) or ssarv_analysis (production integration).
add_library(ssarv_allocators src/allocators.cpp)
target_link_libraries(ssarv_allocators PUBLIC ssarv_utilities)
add_library(ssarv_phi src/phielim.cpp)
target_link_libraries(ssarv_phi PUBLIC ssarv_utilities)
add_library(ssarv_lowering src/lowering.cpp)
target_link_libraries(ssarv_lowering PUBLIC ssarv_utilities)
add_library(ssarv_lead src/parser.cpp src/chordality.cpp src/advanced_allocators.cpp
    src/ilp.cpp src/coalesce.cpp src/verify.cpp src/backend.cpp src/driver.cpp)
target_link_libraries(ssarv_lead PUBLIC ssarv_allocators ssarv_phi ssarv_lowering ssarv_analysis)
add_executable(ssarv src/main.cpp)
target_link_libraries(ssarv PRIVATE ssarv_lead)

include(CTest)
if(BUILD_TESTING)
    find_package(GTest REQUIRED)
    function(contract_target target)
        add_executable(${target} ${ARGN})
        target_include_directories(${target} PRIVATE tests)
        target_link_libraries(${target} PRIVATE GTest::gtest_main)
    endfunction()
    function(contract_check name target switch labels)
        option(${switch} "Enable ${name} acceptance checks" OFF)
        add_test(NAME ${name} COMMAND ${target})
        set_tests_properties(${name} PROPERTIES LABELS "${labels}")
        if(NOT ${switch})
            set_tests_properties(${name} PROPERTIES DISABLED TRUE)
        endif()
    endfunction()
    contract_target(scaffold-tests tests/test_scaffold.cpp)
    target_link_libraries(scaffold-tests PRIVATE ssarv_lead)
    add_test(NAME scaffold COMMAND scaffold-tests)
    set_tests_properties(scaffold PROPERTIES LABELS scaffold)

    contract_target(utilities-tests tests/test_utilities.cpp)
    target_link_libraries(utilities-tests PRIVATE ssarv_utilities)
    contract_check(contract.utilities utilities-tests SSARV_TEST_UTILITIES "utilities;group4")

    contract_target(analysis-tests tests/test_analysis.cpp)
    target_link_libraries(analysis-tests PRIVATE ssarv_analysis)
    contract_check(contract.analysis analysis-tests SSARV_TEST_ANALYSIS "group1")

    contract_target(allocators-tests tests/test_allocators.cpp tests/support/analysis_provider.cpp)
    target_link_libraries(allocators-tests PRIVATE ssarv_allocators)
    contract_check(contract.allocators allocators-tests SSARV_TEST_ALLOCATORS "group2")

    contract_target(phi-tests tests/test_phi.cpp)
    target_link_libraries(phi-tests PRIVATE ssarv_phi)
    contract_check(contract.phi phi-tests SSARV_TEST_PHI "group3")
    contract_target(lowering-tests tests/test_lowering.cpp)
    target_link_libraries(lowering-tests PRIVATE ssarv_lowering)
    contract_check(contract.lowering lowering-tests SSARV_TEST_LOWERING "group3")

    contract_target(lead-tests tests/test_lead.cpp)
    target_link_libraries(lead-tests PRIVATE ssarv_lead)
    contract_check(contract.lead lead-tests SSARV_TEST_LEAD "group4")
    contract_target(integration-tests tests/test_integration.cpp)
    target_link_libraries(integration-tests PRIVATE ssarv_lead)
    contract_check(contract.integration integration-tests SSARV_TEST_INTEGRATION "integration")
endif()
